services:
  web:
    image: demo/web:2.14.3
    ports:
      - "8080:3000"
    environment:
      NODE_ENV: production
      DB_HOST: db
      DB_PASSWORD: ${DB_PASSWORD}
      API_TOKEN: ${API_TOKEN:?missing}
      REDIS_URL: redis://cache:6379
    secrets:
      - db_password
      - api_token
    depends_on:
      - db
  db:
    image: postgres:16
    environment:
      POSTGRES_USER: demo
      POSTGRES_PASSWORD_FILE: /run/secrets/db_password
    secrets:
      - db_password
secrets:
  db_password:
    file: ./secrets/db_password.txt
  api_token:
    external: true

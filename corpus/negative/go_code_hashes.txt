package checksum

import (
	"crypto/sha256"
	"encoding/hex"
	"testing"
)

// Test vectors for the digest helper. Each sha256 digest is of the input string.
var vectors = []struct {
	input  string
	sha256 string
}{
	{"", "e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855"},
	{"abc", "ba7816bf8f01cfea414140de5dae2223b00361a396177a9cb410ff61f20015ad"},
	{"hello world", "b94d27b9934d3e08a52e52d7da7dabfac484efe37a5380ee9088f7ace2efcde9"},
	{"demo-service", "a06ae7aaa21885b14f3c33a08be9f1ab02dbe79a552f2eb45fd0e37c658a8e25"},
	{"release-2.14.3", "b43e1352875658b782dd51902a920bef9d6d4096c4c9763208f567bcf3216ee8"},
	{"The quick brown fox", "5cac4f980fedc3d3f1f99b4be3472c9b30d56523e632d151237ec9309048bda9"},
}

const emptyDigest = "e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855" // sha256 of empty input
const releaseChecksum = "68b076be5ba974795f927bf1aa53b0dc2991b90d8d535a9dbaf33bad74a2d3a3" // checksum of demo-service-2.14.3.tar.gz

func TestDigest(t *testing.T) {
	for _, v := range vectors {
		sum := sha256.Sum256([]byte(v.input))
		if hex.EncodeToString(sum[:]) != v.sha256 {
			t.Errorf("digest mismatch for %q", v.input)
		}
	}
}
